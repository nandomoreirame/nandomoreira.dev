# Plano: i18n-content

> **Spec:** docs/specs/i18n-content/SPEC.md

**Objetivo:** traduzir para o inglês os 26 cases e os 22 posts não arquivados, servir o conteúdo inglês nas rotas de máquina e ligar `EN_LIVE`.
**Arquitetura:** coleções de sobreposição (`postsEn`, `projectsEn`) com `translationOf` e só os campos traduzíveis. A camada de acesso valida e funde cada sobreposição com a fonte pt-BR (funções puras em `src/lib/content/translations.ts`), aplica a redação de confidencialidade e substitui os retornos vazios deixados pela fundação. Portões de paridade, de travessão, de português residual e de cobertura controlam a qualidade, porque não há revisão humana antes do push (C5).
**Stack:** Astro 7.2 Content Collections, Zod 4, Vitest.
**Total de tarefas:** 26, em 6 lotes.
**Complexidade estimada:** large.
**Depende de:** `i18n-foundation` implementada (camada de acesso com `locale`, `translationPairs`, `ptLeftovers`, `EN_LIVE`, rotas `/en/portfolio/[slug]`, `/en/lab/[slug]` e `/en/[slug]`).

## Regras para quem implementa

- DO NOT run any git commands (commit, push, branch, merge, add). DO NOT invoke the git-commit skill.
- TDD: as tarefas de código começam pelo teste vermelho. Nas de tradução, o "teste vermelho" é o portão de paridade da tarefa 5 aplicado ao arquivo novo, mais o teste de render da tarefa.
- **Toda tradução segue `docs/specs/i18n-content/TRANSLATION-GUIDE.md` (tarefa 7). Leia o guia inteiro antes da primeira tradução e releia o glossário antes de cada lote.**
- Pare o dev server antes de criar arquivos nas coleções novas (`./node_modules/.bin/astro dev stop`) e suba de novo depois: arquivo novo não entra no store de um dev já rodando (armadilha do `CLAUDE.md`).
- Nunca copie `clientName` nem `confidentialMetrics` de case confidencial para fora do fonte; o arquivo inglês só traduz `confidentialMetrics`, que continua sendo removido na leitura pública.
- No fim de cada tarefa: o teste da tarefa verde, `npm test` sem falha nova e `npm run check` sem erro novo.

`src/content/schemas/post-translation.ts`:

```ts
import { z } from 'zod';

/**
 * English overlay of a post (i18n-content C1, C3). Only what is translated
 * lives here; date, draft, archived, art and the cover image come from the
 * pt-BR source in the access layer, so they cannot drift. `.strict()` makes a
 * copied structural field a build error instead of a silent duplicate.
 */
export const postTranslationSchema = z
  .object({
    slug: z
      .string()
      .min(1)
      .regex(/^[a-z0-9]+(-[a-z0-9]+)*$/, 'slug must be kebab-case')
      .refine((slug) => slug !== 'ia' && slug !== 'en', 'slug "ia" and "en" are reserved'),
    /** pt-BR slug of the source post. */
    translationOf: z.string().min(1),
    title: z.string().min(1),
    seoTitle: z.string().min(1).max(60).optional(),
    summary: z.string().min(1),
    markers: z.array(z.string().min(1)).min(3),
    /** Alt text of the source cover; required when the source has one (REQ-004). */
    coverAlt: z.string().min(1).optional(),
    extras: z.array(z.string().min(1).max(60)).min(1).max(2).optional(),
    keyPoints: z.array(z.string().min(1).max(140)).min(1).max(3).optional(),
    snapshot: z
      .object({
        version: z.string().min(1).max(16).optional(),
        items: z
          .array(z.object({ label: z.string().min(1).max(20), value: z.string().min(1).max(32) }))
          .min(1)
          .max(6),
      })
      .optional(),
  })
  .strict();

export type PostTranslation = z.infer<typeof postTranslationSchema>;
```

 * flag come from the pt-BR source; the client name never leaves it.

export const projectTranslationSchema = z

    translationOf: z.string().min(1),

    title: z.string().min(1).optional(),

    problem: z.string().min(1),

```ts
import type { Post } from '../../content/schemas/post';
import type { PostTranslation } from '../../content/schemas/post-translation';
import type { Project } from '../../content/schemas/project';
import type { ProjectTranslation } from '../../content/schemas/project-translation';

/**
 * Validation and merge of the English overlays (i18n-content C1). Pure: the
 * access layer reads the collections and calls these, so the rules are tested
 * without Astro. A failure throws, which fails the build (REQ-001 to REQ-004).
 */

const fail = (message: string): never => {
  throw new Error(`[i18n-content] ${message}`);
};

const assertUnique = (values: readonly string[], what: (value: string) => string): void => {
  const seen = new Set<string>();
  for (const value of values) {
    if (seen.has(value)) fail(what(value));
    seen.add(value);
  }
};

export const assertPostTranslations = (sources: readonly Post[], overlays: readonly PostTranslation[]): void => {
  const bySlug = new Map(sources.map((post) => [post.slug, post]));

  for (const overlay of overlays) {
    const source = bySlug.get(overlay.translationOf);
    if (!source) fail(`post ${overlay.slug} translates ${overlay.translationOf}, which does not exist`);
    if (source!.archived) fail(`post ${overlay.slug} translates ${overlay.translationOf}, which is archived`);
    if (source!.cover && !overlay.coverAlt) fail(`post ${overlay.slug} needs coverAlt: its source has a cover`);
    if (bySlug.has(overlay.slug)) fail(`post slug ${overlay.slug} collides with a pt-BR post`);
  }

  assertUnique(overlays.map((overlay) => overlay.translationOf), (slug) => `post ${slug} is translated twice`);
  assertUnique(overlays.map((overlay) => overlay.slug), (slug) => `English post slug ${slug} is used twice`);
};

export const mergePost = (source: Post, overlay: PostTranslation): Post => ({
  ...source,
  slug: overlay.slug,
  title: overlay.title,
  seoTitle: overlay.seoTitle,
  summary: overlay.summary,
  markers: overlay.markers,
  extras: overlay.extras,
  keyPoints: overlay.keyPoints,
  snapshot: overlay.snapshot,
  cover: source.cover ? { ...source.cover, alt: overlay.coverAlt ?? source.cover.alt } : undefined,
});

export const assertProjectTranslations = (
  sources: readonly Project[],
  overlays: readonly ProjectTranslation[],
): void => {
  const bySlug = new Map(sources.map((project) => [project.slug, project]));

  for (const overlay of overlays) {
    const source = bySlug.get(overlay.translationOf);
    if (!source) fail(`case translation points at ${overlay.translationOf}, which does not exist`);
    if ((source!.images?.length ?? 0) !== (overlay.imageAlts?.length ?? 0)) {
      fail(`case ${overlay.translationOf}: imageAlts must have one entry per source image`);
    }
    if ((source!.shortcuts?.length ?? 0) !== (overlay.shortcutActions?.length ?? 0)) {
      fail(`case ${overlay.translationOf}: shortcutActions must have one entry per source shortcut`);
    }
  }

  assertUnique(overlays.map((overlay) => overlay.translationOf), (slug) => `case ${slug} is translated twice`);
};

export const mergeProject = (source: Project, overlay: ProjectTranslation): Project => ({
  ...source,
  title: overlay.title ?? source.title,
  summary: overlay.summary,
  problem: overlay.problem,
  decisions: overlay.decisions,
  outcome: overlay.outcome,
  images: source.images?.map((image, index) => ({ ...image, alt: overlay.imageAlts?.[index] ?? image.alt })),
  shortcuts: source.shortcuts?.map((shortcut, index) => ({
    ...shortcut,
    action: overlay.shortcutActions?.[index] ?? shortcut.action,
  })),
  confidentialMetrics: source.confidentialMetrics ? overlay.confidentialMetrics ?? source.confidentialMetrics : undefined,
});
```

import type { TranslationPair } from '../content';
import { staticCounterpart } from './routes';

const OWN = /^https:\/\/(www\.)?nandomoreira\.dev/;

/** Where an internal link in an English text should point (C4). */
export const englishHref = (href: string, pairs: readonly TranslationPair[]): string => {
  if (/^https?:\/\//.test(href) && !OWN.test(href)) return href;
  const local = href.replace(OWN, '') || '/';
  const [path, hash = ''] = local.split('#');
  const pair = staticCounterpart(path!) ?? pairs.find((item) => item['pt-BR'] === path);
  return pair ? `${pair.en}${hash ? `#${hash}` : ''}` : href;
};

/** Internal link targets of a markdown body, outside code and images. */

    .replace(/`[^`]*`/g, ' ')
    .replace(/!\[[^\]]*\]\([^)]*\)/g, ' ')
    .matchAll(/\[[^\]]*\]\(([^)\s]+)\)/g)]
    .map((match) => match[1]!)
    .filter((href) => href.startsWith('/') || OWN.test(href));

import { join } from 'node:path';

export interface MarkdownFile {
  file: string;
  frontmatter: string;
  body: string;
  field: (name: string) => string | undefined;
}

const read = (path: string): MarkdownFile => {
  const source = readFileSync(path, 'utf8');
  const [, frontmatter = '', ...rest] = source.split(/^---$/m);
  return {
    file: path,
    frontmatter,
    body: rest.join('---'),
    field: (name) => frontmatter.match(new RegExp(`^${name}:\\s*"?([^"\\n]+)"?`, 'm'))?.[1]?.trim(),
  };
};

export const markdownIn = (dir: string): MarkdownFile[] =>
  readdirSync(dir)
    .filter((file) => file.endsWith('.md'))
    .map((file) => read(join(dir, file)));

    const pt = sources.get(en.field('translationOf'));
    if (!pt) throw new Error(`${en.file}: translationOf has no source`);
