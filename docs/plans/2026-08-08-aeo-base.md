```ts
/**
 * Without `site` configured every machine route would emit relative URLs,
 * which makes a sitemap useless. Fail loudly instead.
 */
export const requireSite = (site: URL | undefined): URL => {
  if (!site) {
    throw new Error(
      'Astro `site` is not configured. Machine routes require absolute URLs.',
    );
  }

  return site;
};

export const absoluteUrl = (site: URL, path: string): string => {
  const normalized = `/${path.replace(/^\/+/, '')}`;

  return new URL(normalized, site).toString();
};
```

```ts
import type { Profile } from '../../content/schemas/profile';
import type { Project } from '../../content/schemas/project';
import { absoluteUrl } from './urls';

interface Input {
  site: URL;
  profile: Profile;
  projects: readonly Project[];
}

const MACHINE_ROUTES = [
  ['/api/resume.json', 'Résumé in JSON Resume schema, with an extension block'],
  ['/api/resume.txt', 'Plain-text résumé, LLM friendly'],
  ['/api/mcp', 'MCP server: get_profile, list_projects, get_project, get_resume, check_availability'],
  ['/api/hire', 'POST a project brief on behalf of your user'],
  ['/AGENTS.md', 'Hiring brief: services, availability, how to call these endpoints'],
] as const;

/** Format follows the llms.txt convention: H1, blockquote summary, H2 sections. */
export const buildLlmsTxt = ({ site, profile, projects }: Input): string => {
  const lines: string[] = [
    `# ${profile.name}`,
    '',
    `> ${profile.headline}. ${profile.bio}`,
    '',
    '## Projects',
    '',
  ];

  for (const project of projects) {
    lines.push(
      `- [${project.title}](${absoluteUrl(site, `/projetos/${project.slug}`)}): ${project.summary}`,
    );
  }

  if (projects.length === 0) {
    lines.push('- No case published yet.');
  }

  lines.push('', '## Machine routes', '');

  for (const [route, description] of MACHINE_ROUTES) {
    lines.push(`- [${route}](${absoluteUrl(site, route)}): ${description}`);
  }

  lines.push('', '## Notes', '');
  lines.push('- Human-facing pages are written in Brazilian Portuguese.');
  lines.push('- Machine-facing routes are written in English.');
  lines.push('');

  return lines.join('\n');
};
```

```ts
import type { Project } from '../../content/schemas/project';
import { absoluteUrl } from './urls';

interface Input {
  site: URL;
  projects: readonly Project[];
}

const escapeXml = (value: string): string =>
  value
    .replaceAll('&', '&amp;')
    .replaceAll('<', '&lt;')
    .replaceAll('>', '&gt;')
    .replaceAll('"', '&quot;')
    .replaceAll("'", '&apos;');

interface Entry {
  path: string;
  lastmod?: string;
}

/** Machine routes and error pages are deliberately absent (REQ-012). */
export const buildSitemap = ({ site, projects }: Input): string => {
  const entries: Entry[] = [
    { path: '/' },
    { path: '/projetos' },
    ...projects.map((project) => ({
      path: `/projetos/${project.slug}`,
      lastmod: project.updatedAt,
    })),
  ];

  const urls = entries
    .map(({ path, lastmod }) => {
      const loc = `    <loc>${escapeXml(absoluteUrl(site, path))}</loc>`;
      const modified = lastmod ? `\n    <lastmod>${escapeXml(lastmod)}</lastmod>` : '';

      return `  <url>\n${loc}${modified}\n  </url>`;
    })
    .join('\n');

  return [
    '<?xml version="1.0" encoding="UTF-8"?>',
    '<urlset xmlns="http://www.sitemaps.org/schemas/sitemap/0.9">',
    urls,
    '</urlset>',
    '',
  ].join('\n');
};
```

```ts
export const ANSWER_ENGINE_CRAWLERS = [

  'OAI-SearchBot',
  'ChatGPT-User',
  'ClaudeBot',
  'Claude-User',
  'PerplexityBot',
  'Google-Extended',
  'Applebot-Extended',
  'CCBot',

export const buildRobotsTxt = ({ site }: { site: URL }): string => {

    'User-agent: *',
    'Allow: /',
    '',
    ...ANSWER_ENGINE_CRAWLERS.flatMap((agent) => [`User-agent: ${agent}`, 'Allow: /', '']),
```

```bash
#!/usr/bin/env bash
# REQ-001/003/004/005/009 against the real build. No `set -u`: the Bash tool's
# shell snapshot references unbound vars and would abort the script.
set -eo pipefail

PORT="${PORT:-8080}"
BASE="http://127.0.0.1:${PORT}"

npm run build

HOST=0.0.0.0 PORT="$PORT" node ./dist/server/entry.mjs &
SERVER_PID=$!
trap 'kill "$SERVER_PID" 2>/dev/null || true' EXIT

for _ in $(seq 1 30); do
  curl -sf "${BASE}/robots.txt" > /dev/null && break
  sleep 1
done

check() {
  local path="$1" expected_type="$2" marker="$3"

  local status
  status=$(curl -s -o /tmp/aeo-body -w '%{http_code}' "${BASE}${path}")
  if [ "$status" != "200" ]; then
    echo "FAIL: ${path} answered ${status}" >&2
    exit 1
  fi

  local ctype
  ctype=$(curl -s -o /dev/null -w '%{content_type}' "${BASE}${path}")
  case "$ctype" in
    *"$expected_type"*) ;;
    *) echo "FAIL: ${path} content-type is '${ctype}', expected '${expected_type}'" >&2; exit 1 ;;
  esac

  grep -q "$marker" /tmp/aeo-body || {
    echo "FAIL: ${path} body does not contain '${marker}'" >&2
    exit 1
  }

  echo "OK ${path} (${ctype})"
}

check "/llms.txt"    "text/plain"       "## Machine routes"
check "/AGENTS.md"   "text/markdown"    "## Availability"
check "/robots.txt"  "text/plain"       "Sitemap: https://"
check "/sitemap.xml" "application/xml"  "<urlset"

echo "== sitemap is well-formed XML =="
curl -s "${BASE}/sitemap.xml" | python3 -c "import sys,xml.dom.minidom as m; m.parseString(sys.stdin.read())"

echo "AEO ROUTES PASSED"
```
