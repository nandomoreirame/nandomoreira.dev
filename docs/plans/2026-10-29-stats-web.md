```rust
//! Calendar slices for the web panel (stats-web REQ-007 to REQ-009): a day,
//! a Monday-to-Sunday week, a whole month, the rolling 7 and 30 days or a
//! free range, all São Paulo dates and never past today.

use crate::period::sao_paulo;
use chrono::{DateTime, Datelike, Duration, NaiveDate, Utc};

/// The daily counters live 24 months (DAILY_TTL_MONTHS in src/lib/analytics/event.ts).
pub const MAX_DAYS: i64 = 731;

const MONTHS: [&str; 12] = [
    "janeiro", "fevereiro", "março", "abril", "maio", "junho", "julho", "agosto", "setembro",
    "outubro", "novembro", "dezembro",
];

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Span {
    Day,
    Week,
    Month,
    Last7,
    Last30,
    Custom,
}

const SPANS: [Span; 6] = [Span::Day, Span::Week, Span::Month, Span::Last7, Span::Last30, Span::Custom];

impl Span {
    pub fn slug(self) -> &'static str {
        match self {
            Span::Day => "day",
            Span::Week => "week",
            Span::Month => "month",
            Span::Last7 => "last7",
            Span::Last30 => "last30",
            Span::Custom => "custom",
        }
    }

    pub fn from_slug(slug: &str) -> Option<Span> {
        SPANS.into_iter().find(|span| span.slug() == slug)
    }

    /// Day, week and month step with ← →; the rolling and free ranges do not.
    pub fn is_calendar(self) -> bool {
        matches!(self, Span::Day | Span::Week | Span::Month)
    }
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Range {
    pub span: Span,
    pub from: NaiveDate,
    pub to: NaiveDate,
}

/// Today in São Paulo, as a date.
pub fn today_date(now: DateTime<Utc>) -> NaiveDate {
    now.with_timezone(&sao_paulo()).date_naive()
}

fn last_of_month(first: NaiveDate) -> NaiveDate {
    let (year, month) = if first.month() == 12 {
        (first.year() + 1, 1)
    } else {
        (first.year(), first.month() + 1)
    };
    NaiveDate::from_ymd_opt(year, month, 1)
        .and_then(|next| next.pred_opt())
        .unwrap_or(first)
}

fn br(day: NaiveDate) -> String {
    day.format("%d/%m/%Y").to_string()
}

impl Range {
    /// The slice of `span` that holds `anchor`, cut at today. A free range
    /// comes from `custom`; here it reads as the anchor day.
    pub fn around(span: Span, anchor: NaiveDate, today: NaiveDate) -> Range {
        let anchor = anchor.min(today);
        let (from, to) = match span {
            Span::Day | Span::Custom => (anchor, anchor),
            Span::Week => {
                let from = anchor - Duration::days(anchor.weekday().num_days_from_monday().into());
                (from, from + Duration::days(6))
            }
            Span::Month => {
                let from = anchor.with_day(1).unwrap_or(anchor);
                (from, last_of_month(from))
            }
            Span::Last7 => (today - Duration::days(6), today),
            Span::Last30 => (today - Duration::days(29), today),
        };
        Range {
            span,
            from,
            to: to.min(today),
        }
    }

    /// REQ-009: start on or before the end, the end not past today, at most 24 months.
    pub fn custom(from: NaiveDate, to: NaiveDate, today: NaiveDate) -> Result<Range, String> {
        if from > to {
            return Err(format!("o início ({}) vem depois do fim ({})", br(from), br(to)));
        }
        if to > today {
            return Err(format!("o fim ({}) está no futuro", br(to)));
        }
        if (to - from).num_days() + 1 > MAX_DAYS {
            return Err("o intervalo passa de 24 meses, o limite dos contadores diários".to_owned());
        }
        Ok(Range {
            span: Span::Custom,
            from,
            to,
        })
    }

    pub fn previous(self, today: NaiveDate) -> Option<Range> {
        if !self.span.is_calendar() {
            return None;
        }
        self.from
            .pred_opt()
            .map(|day| Range::around(self.span, day, today))
    }

    /// The next slice, or None once it would start after today.
    pub fn next(self, today: NaiveDate) -> Option<Range> {
        if !self.span.is_calendar() {
            return None;
        }
        let full_end = match self.span {
            Span::Week => self.from + Duration::days(6),
            Span::Month => last_of_month(self.from),
            _ => self.from,
        };
        full_end
            .succ_opt()
            .filter(|day| *day <= today)
            .map(|day| Range::around(self.span, day, today))
    }

    /// `AAAA-MM-DD` of every day in the slice, oldest first.
    pub fn days(self) -> Vec<String> {
        self.from
            .iter_days()
            .take_while(|day| *day <= self.to)
            .map(|day| day.format("%Y-%m-%d").to_string())
            .collect()
    }

    pub fn label(self) -> String {
        match self.span {
            Span::Day => br(self.from),
            Span::Week => format!("semana de {}", br(self.from)),
            Span::Month => format!("{} de {}", MONTHS[self.from.month0() as usize], self.from.year()),
            Span::Last7 => "últimos 7 dias".to_owned(),
            Span::Last30 => "últimos 30 dias".to_owned(),
            Span::Custom => format!("{} a {}", br(self.from), br(self.to)),
        }
    }
}
```

```rust
//! What one page of the web panel shows, read from and written back to the
//! query string (stats-web REQ-010): tab, slice and the optional channel or
//! city filter. Anything malformed is an error for a 400, never a guess.

use crate::range::{Range, Span};
use chrono::NaiveDate;
use std::collections::BTreeMap;

/// Longest channel or city accepted from the URL.
const MAX_FILTER: usize = 80;

#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
pub enum Tab {
    #[default]
    Live,
    Funnel,
    Traffic,
    Mcp,
    Sales,
}

pub const TABS: [Tab; 5] = [Tab::Live, Tab::Funnel, Tab::Traffic, Tab::Mcp, Tab::Sales];

impl Tab {
    pub fn slug(self) -> &'static str {
        match self {
            Tab::Live => "live",
            Tab::Funnel => "funnel",
            Tab::Traffic => "traffic",
            Tab::Mcp => "mcp",
            Tab::Sales => "sales",
        }
    }

    pub fn title(self) -> &'static str {
        match self {
            Tab::Live => "Ao vivo",
            Tab::Funnel => "Funil",
            Tab::Traffic => "Tráfego",
            Tab::Mcp => "MCP",
            Tab::Sales => "Vendas",
        }
    }

    pub fn from_slug(slug: &str) -> Option<Tab> {
        TABS.into_iter().find(|tab| tab.slug() == slug)
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct View {
    pub tab: Tab,
    pub range: Range,
    /// A past calendar slice keeps its date in the URL; the current one does
    /// not, so "today" and "this week" move on at midnight.
    pub pinned: bool,
    pub channel: Option<String>,
    pub city: Option<String>,
}

fn date(params: &BTreeMap<String, String>, key: &str) -> Result<Option<NaiveDate>, String> {
    match params.get(key).map(String::as_str) {
        None | Some("") => Ok(None),
        Some(text) => NaiveDate::parse_from_str(text, "%Y-%m-%d")
            .map(Some)
            .map_err(|_| format!("data inválida em {key}: use AAAA-MM-DD")),
    }
}

fn filter(params: &BTreeMap<String, String>, key: &str) -> Result<Option<String>, String> {
    match params.get(key).map(|value| value.trim()) {
        None | Some("") => Ok(None),
        Some(value)
            if value.chars().count() > MAX_FILTER || value.chars().any(char::is_control) =>
        {
            Err(format!("filtro inválido em {key}"))
        }
        Some(value) => Ok(Some(value.to_owned())),
    }
}

fn iso(day: NaiveDate) -> String {
    day.format("%Y-%m-%d").to_string()
}

impl View {
    pub fn today(today: NaiveDate) -> View {
        View {
            tab: Tab::Live,
            range: Range::around(Span::Day, today, today),
            pinned: false,
            channel: None,
            city: None,
        }
    }

    pub fn parse(params: &BTreeMap<String, String>, today: NaiveDate) -> Result<View, String> {
        let tab = match params.get("tab") {
            None => Tab::Live,
            Some(slug) => Tab::from_slug(slug).ok_or_else(|| format!("aba desconhecida: {slug}"))?,
        };
        let span = match params.get("span") {
            None => Span::Day,
            Some(slug) => {
                Span::from_slug(slug).ok_or_else(|| format!("período desconhecido: {slug}"))?
            }
        };
        let range = if span == Span::Custom {
            let (Some(from), Some(to)) = (date(params, "from")?, date(params, "to")?) else {
                return Err("intervalo livre precisa de from e to".to_owned());
            };
            Range::custom(from, to, today)?
        } else {
            let at = date(params, "at")?.unwrap_or(today);
            if at > today {
                return Err(format!("{} está no futuro", at.format("%d/%m/%Y")));
            }
            Range::around(span, at, today)
        };
        Ok(View {
            tab,
            range,
            pinned: range.span.is_calendar() && range.to < today,
            channel: filter(params, "channel")?,
            city: filter(params, "city")?,
        })
    }

    /// The query string that reproduces this view, keys in a fixed order.
    pub fn query(&self) -> String {
        let mut query = form_urlencoded::Serializer::new(String::new());
        query.append_pair("tab", self.tab.slug());
        query.append_pair("span", self.range.span.slug());
        if self.range.span == Span::Custom {
            query.append_pair("from", &iso(self.range.from));
            query.append_pair("to", &iso(self.range.to));
        } else if self.pinned {
            query.append_pair("at", &iso(self.range.from));
        }
        if let Some(channel) = &self.channel {
            query.append_pair("channel", channel);
        }
        if let Some(city) = &self.city {
            query.append_pair("city", city);
        }
        query.finish()
    }

    pub fn href(&self) -> String {
        format!("/?{}", self.query())
```

```rust
//! The raw events a view asks for (stats-web REQ-011, REQ-013): the channel
//! or city filter, and how much of the slice the 30 days of raw events still
//! cover.

use crate::model::Event;
use crate::period::Period;
use crate::range::Range;
use chrono::{DateTime, Utc};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Coverage {
    Full,
    Partial,
    None,
}

pub fn coverage(range: &Range, now: DateTime<Utc>) -> Coverage {
    let window = Period::Month.days(now);
    let days = range.days();
    let inside = days.iter().filter(|day| window.contains(day)).count();
    if inside == days.len() {
        Coverage::Full
    } else if inside == 0 {
        Coverage::None
    } else {
        Coverage::Partial
    }
}

/// The events of `channel` and of the place `city` ("City, ST", as
/// `Event::place` prints it in the ranking); None keeps every event.
pub fn filter_events(events: &[Event], channel: Option<&str>, city: Option<&str>) -> Vec<Event> {
    events
        .iter()
        .filter(|event| channel.is_none_or(|channel| event.channel == channel))
        .filter(|event| city.is_none_or(|city| event.place().as_deref() == Some(city)))
        .cloned()
        .collect()
}
```

```rust
//! The daily series behind the web chart (stats-web REQ-012): people's page
//! views, agents and leads per day from the daily counters, folded into
//! weeks when the slice is too long to read day by day.

use crate::model::Daily;
use chrono::{Datelike, Duration, NaiveDate};
use std::collections::BTreeMap;

/// Above this many days the chart shows weeks; the table keeps the days.
pub const WEEKLY_ABOVE: usize = 120;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Point {
    pub start: String,
    pub label: String,
    pub humans: i64,
    pub agents: i64,
    pub leads: i64,
}

fn short(day: NaiveDate) -> String {
    day.format("%d/%m").to_string()
}

/// One point per day of `days`; a day with no counter document counts 0.
pub fn daily_points(days: &[String], dailies: &BTreeMap<String, Daily>) -> Vec<Point> {
    days.iter()
        .filter_map(|day| {
            let date: NaiveDate = day.parse().ok()?;
            let counters = dailies.get(day);
            let get = |group: &str, key: &str| counters.map_or(0, |d| d.get(group, key));
            Some(Point {
                start: day.clone(),
                label: short(date),
                humans: get("traffic", "human"),
                agents: get("traffic", "ai_agent"),
                leads: get("conversions", "cv") + get("conversions", "hire"),
            })
        })
        .collect()
}

/// The points to draw: the days, or Monday-started weeks above WEEKLY_ABOVE.
pub fn chart_points(points: &[Point]) -> Vec<Point> {
    if points.len() <= WEEKLY_ABOVE {
        return points.to_vec();
    }
    let mut weeks: Vec<Point> = Vec::new();
    for point in points {
        let Ok(date) = point.start.parse::<NaiveDate>() else {
            continue;
        };
        let monday = date - Duration::days(date.weekday().num_days_from_monday().into());
        let key = monday.format("%Y-%m-%d").to_string();
        match weeks.last_mut() {
            Some(week) if week.start == key => {
                week.humans += point.humans;
                week.agents += point.agents;
                week.leads += point.leads;
            }
            _ => weeks.push(Point {
                start: key,
                label: format!("sem. {}", short(monday)),
                humans: point.humans,
                agents: point.agents,
                leads: point.leads,
            }),
        }
    }
    weeks
}
```

```rust
//! The series as an inline SVG line chart (stats-web REQ-012): built on the
//! server without a library, coloured by CSS classes so both themes apply.
//! Text inside is escaped here, because the template prints it as `safe`.

use crate::series::Point;
use std::fmt::Write;

const WIDTH: f64 = 720.0;
const HEIGHT: f64 = 220.0;
const PAD: f64 = 24.0;

fn escape(text: &str) -> String {
    text.replace('&', "&amp;")
        .replace('<', "&lt;")
        .replace('>', "&gt;")
        .replace('"', "&quot;")
}

pub fn series_svg(points: &[Point], title: &str) -> String {
    let (Some(first), Some(last)) = (points.first(), points.last()) else {
        return String::new();
    };
    let max = points
        .iter()
        .flat_map(|p| [p.humans, p.agents, p.leads])
        .max()
        .unwrap_or(0)
        .max(1) as f64;
    let step = if points.len() > 1 {
        (WIDTH - 2.0 * PAD) / (points.len() - 1) as f64
    } else {
        0.0
    };
    let x = |i: usize| PAD + step * i as f64;
    let y = |value: i64| HEIGHT - PAD - (HEIGHT - 2.0 * PAD) * (value as f64 / max);
    let series: [(&str, fn(&Point) -> i64); 3] = [
        ("chart-humans", |p| p.humans),
        ("chart-agents", |p| p.agents),
        ("chart-leads", |p| p.leads),
    ];

    let mut svg = String::new();
    let _ = write!(
        svg,
        r#"<svg class="chart" viewBox="0 0 {WIDTH} {HEIGHT}" role="img" aria-labelledby="chart-title" preserveAspectRatio="none"><title id="chart-title">{}</title>"#,
        escape(title)
    );
    let _ = write!(
        svg,
        r#"<line class="chart-axis" x1="{PAD}" y1="{base:.1}" x2="{right:.1}" y2="{base:.1}"/>"#,
        base = HEIGHT - PAD,
        right = WIDTH - PAD
    );
    for (class, pick) in series {
        let line: Vec<String> = points
            .iter()
            .enumerate()
            .map(|(i, p)| format!("{:.1},{:.1}", x(i), y(pick(p))))
            .collect();
        let _ = write!(svg, r#"<polyline class="{class}" fill="none" points="{}"/>"#, line.join(" "));
    }
    let _ = write!(
        svg,
        r#"<text class="chart-label" x="{PAD}" y="{HEIGHT}">{}</text><text class="chart-label" x="{right:.1}" y="{HEIGHT}" text-anchor="end">{}</text></svg>"#,
        escape(&first.label),
        escape(&last.label),
        right = WIDTH - PAD
    );
    svg
}
```

```rust
pub fn dailies_path(events: &Path) -> PathBuf {

}

fn counters_of(value: &Value) -> BTreeMap<String, i64> {
    value
        .as_object()
        .into_iter()
        .flatten()
        .filter_map(|(key, n)| Some((key.clone(), n.as_i64()?)))
        .collect()
}

fn decode_dailies(text: &str) -> BTreeMap<String, Daily> {
    let Ok(root) = serde_json::from_str::<Value>(text) else {
        return BTreeMap::new();
    };
    if root.get("version").and_then(Value::as_u64) != Some(VERSION) {
        return BTreeMap::new();
    }
    root.get("days")
        .and_then(Value::as_object)
        .into_iter()
        .flatten()
        .map(|(day, groups)| {
            let groups = groups
                .as_object()
                .into_iter()
                .flatten()
                .map(|(group, counters)| (group.clone(), counters_of(counters)))
                .collect();
            (day.clone(), Daily { day: day.clone(), groups })
        })
        .collect()
}

/// The daily counters of days that already ended (stats-web REQ-019): they
/// never change again, so the web panel reads each one from Firestore once.
#[derive(Debug, Default)]
pub struct DailyCache {
    path: Option<PathBuf>,
    days: BTreeMap<String, Daily>,
    dirty: bool,
}

impl DailyCache {
    pub fn open(path: Option<PathBuf>) -> DailyCache {
        let days = path
            .as_deref()
            .and_then(|path| std::fs::read_to_string(path).ok())
            .map(|text| decode_dailies(&text))
            .unwrap_or_default();
        DailyCache { path, days, dirty: false }
    }

    pub fn get(&self, day: &str) -> Option<&Daily> {
        self.days.get(day)
    }

    /// Keeps `daily` only when its day is over: today's counters still move.
    pub fn keep(&mut self, daily: &Daily, today: &str) {
        if daily.day.as_str() < today && self.days.get(&daily.day) != Some(daily) {
            self.days.insert(daily.day.clone(), daily.clone());
            self.dirty = true;
        }
    }

    /// Drops the days before `oldest` and writes the file, when something changed.
    pub fn save(&mut self, oldest: &str) -> std::io::Result<()> {
        let Some(path) = self.path.as_deref() else {
            return Ok(());
        };
        if !self.dirty {
            return Ok(());
        }
        self.days.retain(|day, _| day.as_str() >= oldest);
        let days: serde_json::Map<String, Value> = self
            .days
            .iter()
            .map(|(day, daily)| (day.clone(), json!(daily.groups)))
            .collect();
        write_private(path, &json!({"version": VERSION, "days": days}).to_string())?;
        self.dirty = false;
        Ok(())
    }
}
```

```rust
//! Firestore tokens for the web server (stats-web REQ-003, REQ-004): from a
//! service account key when GOOGLE_APPLICATION_CREDENTIALS points at one,
//! otherwise from gcloud, like the TUI. A key that does not read keeps the
//! server up and every refresh reports why (no restart loop).

use anyhow::{Context, anyhow, bail};
use jsonwebtoken::{Algorithm, EncodingKey, Header};
use serde::Serialize;
use serde_json::Value;
use std::time::{Duration, SystemTime, UNIX_EPOCH};
use tokio::time::Instant;

const SCOPE: &str = "https://www.googleapis.com/auth/datastore";
const GRANT: &str = "urn:ietf:params:oauth:grant-type:jwt-bearer";
/// Google issues one-hour tokens; renew this long before they end.
const MARGIN: Duration = Duration::from_secs(5 * 60);
/// gcloud does not say when its token ends; the TUI assumes 45 minutes.
const GCLOUD_TTL: Duration = Duration::from_secs(45 * 60);

#[derive(Clone, PartialEq, Eq)]
pub struct ServiceAccountKey {
    pub client_email: String,
    pub private_key: String,
    pub token_uri: String,
}

/// Never prints the private key, not even in a panic or a log line.
impl std::fmt::Debug for ServiceAccountKey {
    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
        f.debug_struct("ServiceAccountKey")
            .field("client_email", &self.client_email)
            .finish_non_exhaustive()
    }
}

pub fn parse_key(json: &str) -> anyhow::Result<ServiceAccountKey> {
    let value: Value = serde_json::from_str(json).context("a chave da conta de serviço não é JSON")?;
    if value.get("type").and_then(Value::as_str) != Some("service_account") {
        bail!("o arquivo não é uma chave de conta de serviço");
    }
    let field = |key: &str| {
        value
            .get(key)
            .and_then(Value::as_str)
            .filter(|text| !text.is_empty())
            .map(str::to_owned)
            .ok_or_else(|| anyhow!("chave da conta de serviço sem {key}"))
    };
    Ok(ServiceAccountKey {
        client_email: field("client_email")?,
        private_key: field("private_key")?,
        token_uri: field("token_uri")?,
    })
}

#[derive(Debug, Serialize, PartialEq, Eq)]
pub struct Claims {
    pub iss: String,
    pub scope: String,
    pub aud: String,
    pub iat: u64,
    pub exp: u64,
}

pub fn claims(key: &ServiceAccountKey, now_secs: u64) -> Claims {
    Claims {
        iss: key.client_email.clone(),
        scope: SCOPE.to_owned(),
        aud: key.token_uri.clone(),
        iat: now_secs,
        exp: now_secs + 3600,
    }
}

/// The signed JWT that Google trades for an access token.
pub fn assertion(key: &ServiceAccountKey, now_secs: u64) -> anyhow::Result<String> {
    let pem = EncodingKey::from_rsa_pem(key.private_key.as_bytes())
        .map_err(|_| anyhow!("chave privada da conta de serviço inválida"))?;
    jsonwebtoken::encode(&Header::new(Algorithm::RS256), &claims(key, now_secs), &pem)
        .map_err(|error| anyhow!("não consegui assinar o pedido de token: {error}"))
}

pub fn token_request_body(assertion: &str) -> String {
    form_urlencoded::Serializer::new(String::new())
        .append_pair("grant_type", GRANT)
        .append_pair("assertion", assertion)
        .finish()
}

pub fn parse_token_response(body: &str) -> anyhow::Result<(String, Duration)> {
    let value: Value = serde_json::from_str(body).context("resposta de token ilegível")?;
    let token = value
        .get("access_token")
        .and_then(Value::as_str)
        .ok_or_else(|| anyhow!("o Google não devolveu access_token"))?;
    let ttl = value.get("expires_in").and_then(Value::as_u64).unwrap_or(3600);
    Ok((token.to_owned(), Duration::from_secs(ttl)))
}

pub fn needs_renewal(expires_at: Option<Instant>, now: Instant) -> bool {
    expires_at.is_none_or(|end| now + MARGIN >= end)
}

#[derive(Debug)]
pub enum TokenSource {
    Gcloud,
    ServiceAccount(ServiceAccountKey),
    /// A service account file that did not read: every token request fails with this.
    Broken(String),
}

impl TokenSource {
    /// `contents` is the file GOOGLE_APPLICATION_CREDENTIALS points at, if any.
    /// A user credential (gcloud's ADC) goes through gcloud, as in the TUI.
    pub fn choose(contents: Option<&str>) -> TokenSource {
        let Some(json) = contents else {
            return TokenSource::Gcloud;
        };
        let kind = serde_json::from_str::<Value>(json)
            .ok()
            .and_then(|value| value.get("type")?.as_str().map(str::to_owned));
        if kind.as_deref() != Some("service_account") {
            return TokenSource::Gcloud;
        }
        match parse_key(json) {
            Ok(key) => TokenSource::ServiceAccount(key),
            Err(error) => TokenSource::Broken(error.to_string()),
        }
    }
}

pub struct Tokens {
    source: TokenSource,
    http: reqwest::Client,
    current: Option<(String, Instant)>,
}

impl Tokens {
    pub fn new(source: TokenSource, http: reqwest::Client) -> Tokens {
        Tokens { source, http, current: None }
    }

    /// The token to use now, renewed when it is about to end; `force` after a 401.
    pub async fn token(&mut self, force: bool) -> anyhow::Result<String> {
        let now = Instant::now();
        if !force
            && let Some((token, end)) = &self.current
            && !needs_renewal(Some(*end), now)
        {
            return Ok(token.clone());
        }
        let (token, ttl) = match &self.source {
            TokenSource::Broken(reason) => bail!("credencial inválida: {reason}"),
            TokenSource::Gcloud => {
                let token = tokio::task::spawn_blocking(|| {
                    crate::auth::refresh_token(&crate::auth::System)
                })
                .await??;
                (token, GCLOUD_TTL)
            }
            TokenSource::ServiceAccount(key) => {
                let secs = SystemTime::now().duration_since(UNIX_EPOCH)?.as_secs();
                let response = self
                    .http
                    .post(&key.token_uri)
                    .header("content-type", "application/x-www-form-urlencoded")
                    .body(token_request_body(&assertion(key, secs)?))
                    .send()
                    .await?;
                let status = response.status();
                let body = response.text().await?;
                if !status.is_success() {
                    bail!("credencial inválida: o Google recusou a conta de serviço ({status})");
                }
                parse_token_response(&body)?
            }
        };
        self.current = Some((token.clone(), now + ttl));
        Ok(token)
    }
}
```

```rust
    matches!(result, Err(error) if error.is::<Unauthorized>())

/// Firestore with a token renewed before it ends and once more after a 401.
pub struct Remote {
    firestore: RwLock<Firestore>,
    tokens: Mutex<Tokens>,
}

impl Remote {
    pub fn new(project: &str, tokens: Tokens) -> Remote {
        Remote {
            firestore: RwLock::new(Firestore::new(project, String::new())),
            tokens: Mutex::new(tokens),
        }
    }

    async fn authorize(&self, force: bool) -> anyhow::Result<()> {
        let token = self.tokens.lock().await.token(force).await?;
        self.firestore.write().await.set_token(token);
        Ok(())
    }
}

impl Source for Remote {
    async fn events_since(&self, from: &str) -> anyhow::Result<Vec<Document>> {
        self.authorize(false).await?;
        let first = self.firestore.read().await.events_since(from).await;
        if !is_unauthorized(&first) {
            return first;
        }
        self.authorize(true).await?;
        self.firestore.read().await.events_since(from).await
    }

    async fn dailies(&self, days: &[String]) -> anyhow::Result<Vec<Daily>> {
        self.authorize(false).await?;
```

```rust
//! REQ-002: the panel answers only to names it serves. A page on any other
//! site can point a DNS name at 127.0.0.1 (DNS rebinding); its requests then
//! carry that foreign Host, and get 421 before reaching a route.

use axum::Router;
use axum::extract::{Request, State};
use axum::http::{StatusCode, header};
use axum::middleware::{self, Next};
use axum::response::{IntoResponse, Response};
use std::sync::Arc;

#[derive(Debug, Clone)]
pub struct AllowedHosts(Arc<Vec<String>>);

impl AllowedHosts {
    pub fn new(host_name: Option<&str>) -> AllowedHosts {
        let mut hosts = vec!["localhost".to_owned(), "127.0.0.1".to_owned(), "[::1]".to_owned()];
        if let Some(name) = host_name {
            hosts.push(name.to_ascii_lowercase());
        }
        AllowedHosts(Arc::new(hosts))
    }

    /// The Host header without its port must be one of ours.
    pub fn allows(&self, host: Option<&str>) -> bool {
        let Some(host) = host else {
            return false;
        };
        let name = if host.starts_with('[') {
            host.split_inclusive(']').next().unwrap_or(host)
        } else {
            host.split(':').next().unwrap_or(host)
        };
        self.0.iter().any(|allowed| allowed.eq_ignore_ascii_case(name))
    }
}

/// A form posted from another site carries that site's Origin; ours, or none
/// (same-origin navigation in older browsers), is fine.
pub fn same_origin(origin: Option<&str>, hosts: &AllowedHosts) -> bool {
    match origin {
        None => true,
        Some(origin) => origin
            .strip_prefix("http://")
            .or_else(|| origin.strip_prefix("https://"))
            .is_some_and(|rest| hosts.allows(Some(rest))),
    }
}

async fn guard(State(hosts): State<AllowedHosts>, request: Request, next: Next) -> Response {
    let host = request.headers().get(header::HOST).and_then(|value| value.to_str().ok());
    if hosts.allows(host) {
        next.run(request).await
    } else {
        (StatusCode::MISDIRECTED_REQUEST, "host não permitido").into_response()
    }
}

pub fn guarded(routes: Router, hosts: AllowedHosts) -> Router {
    routes.layer(middleware::from_fn_with_state(hosts, guard))
}
```

```rust
            .iter()
            .rev()
```

```rust
//! The web server's single reader (stats-web REQ-016, REQ-018, REQ-019): one
//! refresh feeds every open page, the daily counters a slice needs are read
//! once, and a failure is kept for the page while the data stays.

use super::store::Store;
use crate::app::days_to_refresh;
use crate::cache::{DailyCache, EventCache};
use crate::model::Daily;
use crate::period::{load_start, today};
use crate::range::MAX_DAYS;
use crate::source::Source;
use chrono::{DateTime, Utc};
use std::time::Duration;
use tokio::sync::{Mutex, Notify, RwLock, broadcast};
use tokio::time::Instant;

/// Days per batchGet when a long slice needs counters.
const DAILY_BATCH: usize = 100;
/// The events file holds about 4 MB; rewrite it at most this often.
const SAVE_EVERY: Duration = Duration::from_secs(10 * 60);

pub struct Hub<S: Source> {
    pub source: S,
    pub store: RwLock<Store>,
    events: Mutex<EventCache>,
    dailies: Mutex<DailyCache>,
    last_day: Mutex<Option<String>>,
    saved_at: Mutex<Option<Instant>>,
    pub(super) tick: broadcast::Sender<()>,
    pub(super) wake: Notify,
}

impl<S: Source> Hub<S> {
    pub fn new(source: S, events: EventCache, dailies: DailyCache) -> Hub<S> {
        let mut store = Store::default();
        store.log.merge(events.documents().to_vec());
        Hub {
            source,
            store: RwLock::new(store),
            events: Mutex::new(events),
            dailies: Mutex::new(dailies),
            last_day: Mutex::new(None),
            saved_at: Mutex::new(None),
            tick: broadcast::channel(16).0,
            wake: Notify::new(),
        }
    }

    /// New events since the newest one held, and today's counters
    /// (yesterday's once more after midnight); then every open page is told.
    pub async fn refresh(&self, now: DateTime<Utc>) {
        let result = self.try_refresh(now).await;
        {
            let mut store = self.store.write().await;
            match result {
                Ok(()) => {
                    store.error = None;
                    store.refreshed_at = Some(now);
                }
                Err(error) => store.error = Some(format!("erro ao ler o Firestore: {error}")),
            }
        }
        let _ = self.tick.send(());
    }

    async fn try_refresh(&self, now: DateTime<Utc>) -> anyhow::Result<()> {
        let from = self.store.read().await.log.cursor().unwrap_or_else(|| load_start(now));
        let fetched = self.source.events_since(&from).await?;
        self.events.lock().await.add(&fetched);
        self.store.write().await.log.merge(fetched);

        let day = today(now);
        let days = days_to_refresh(self.last_day.lock().await.as_deref(), &day);
        let fresh = self.source.dailies(&days).await?;
        {
            let mut cache = self.dailies.lock().await;
            for daily in &fresh {
                cache.keep(daily, &day);
            }
        }
        {
            let mut store = self.store.write().await;
            for daily in fresh {
                store.dailies.insert(daily.day.clone(), daily);
            }
        }
        *self.last_day.lock().await = Some(day);
        Ok(())
    }

    /// REQ-019: the counters of `days` not in memory yet, from the cache or
    /// from Firestore in batches. A day that ended with no document reads as
    /// zero and is not asked for again.
    pub async fn ensure_dailies(&self, days: &[String], now: DateTime<Utc>) {
        let missing: Vec<String> = {
            let store = self.store.read().await;
            days.iter().filter(|day| !store.dailies.contains_key(*day)).cloned().collect()
        };
        if missing.is_empty() {
            return;
        }
        let today = today(now);
        let mut found: Vec<Daily> = Vec::new();
        let mut ask: Vec<String> = Vec::new();
        {
            let cache = self.dailies.lock().await;
            for day in missing {
                match cache.get(&day) {
                    Some(daily) => found.push(daily.clone()),
                    None => ask.push(day),
                }
            }
        }
        for batch in ask.chunks(DAILY_BATCH) {
            match self.source.dailies(batch).await {
                Ok(list) => {
                    let mut cache = self.dailies.lock().await;
                    for day in batch {
                        let daily = list.iter().find(|d| &d.day == day).cloned().or_else(|| {
                            (day.as_str() < today.as_str()).then(|| Daily { day: day.clone(), ..Daily::default() })
                        });
                        if let Some(daily) = daily {
                            cache.keep(&daily, &today);
                            found.push(daily);
                        }
                    }
                }
                Err(error) => {
                    self.store.write().await.error = Some(format!("erro ao ler os contadores diários: {error}"));
                    break;
                }
            }
        }
        let mut store = self.store.write().await;
        for daily in found {
            store.dailies.insert(daily.day.clone(), daily);
        }
    }

    /// Writes both caches once SAVE_EVERY has passed, or always with `force` (on shutdown).
    pub async fn save(&self, now: DateTime<Utc>, force: bool) {
        let instant = Instant::now();
        {
            let mut saved = self.saved_at.lock().await;
            if !force && saved.is_some_and(|at| instant.duration_since(at) < SAVE_EVERY) {
                return;
            }
            *saved = Some(instant);
        }
        let oldest = (now - chrono::Duration::days(MAX_DAYS)).format("%Y-%m-%d").to_string();
        let events = self.events.lock().await.save(&load_start(now));
        let dailies = self.dailies.lock().await.save(&oldest);
        if let Err(error) = events.and(dailies) {
            self.store.write().await.error = Some(format!("erro ao gravar o cache: {error}"));
        }
    }
}
```

```rust
#[derive(Template)]
#[template(path = "page.html")]
pub struct PageTemplate<'a> {
    pub d: &'a Display,
}

#[derive(Template)]
#[template(path = "body.html")]
pub struct BodyTemplate<'a> {
    pub d: &'a Display,
}

#[derive(Template)]
#[template(path = "message.html")]
pub struct MessageTemplate<'a> {
    pub title: &'a str,
    pub message: &'a str,
}

/// No inline script or style anywhere: the bars use classes, the SVG is markup.
const CSP: &str = "default-src 'none'; style-src 'self'; script-src 'self'; connect-src 'self'; img-src 'self'; form-action 'self'; base-uri 'none'; frame-ancestors 'none'";

pub fn render(template: &impl Template) -> Response {
    match template.render() {
        Ok(html) => Html(html).into_response(),
        Err(_) => (StatusCode::INTERNAL_SERVER_ERROR, "erro ao montar a página").into_response(),
    }
}

pub fn message(status: StatusCode, title: &str, text: &str) -> Response {
    (status, render(&MessageTemplate { title, message: text })).into_response()
}

/// The display of `view` from the hub's memory, reading missing counters first.
pub async fn display_of<S: Source>(state: &AppState<S>, view: &View) -> Display {
    let now = Utc::now();
    state.hub.ensure_dailies(&view.range.days(), now).await;
    let store = state.hub.store.read().await;
    build(&snapshot(&store, view, now), view, &store, now, tokio::time::Instant::now())
}

pub async fn page<S: Source>(
    State(state): State<AppState<S>>,
    Query(params): Query<BTreeMap<String, String>>,
) -> Response {
    let view = match View::parse(&params, today_date(Utc::now())) {
        Ok(view) => view,
        Err(error) => return message(StatusCode::BAD_REQUEST, "Pedido inválido", &error),
    };
    let display = display_of(&state, &view).await;
    render(&PageTemplate { d: &display })
}

pub async fn security_headers(request: Request, next: Next) -> Response {
    let mut response = next.run(request).await;
    let headers = response.headers_mut();
    headers.insert(header::CONTENT_SECURITY_POLICY, HeaderValue::from_static(CSP));
    headers.insert(header::X_CONTENT_TYPE_OPTIONS, HeaderValue::from_static("nosniff"));
    headers.insert(header::REFERRER_POLICY, HeaderValue::from_static("no-referrer"));
    headers.insert(header::X_FRAME_OPTIONS, HeaderValue::from_static("DENY"));
    headers.insert("x-robots-tag", HeaderValue::from_static("noindex, nofollow"));
    response
}
```

```rust
//! Live updates (stats-web REQ-016, REQ-017): one poller reads Firestore only
//! while some page listens, every 30 s or every 5 s in the fast mode; each
//! listener gets its own view re-rendered after every refresh, over SSE.

use super::AppState;
use super::host::same_origin;
use super::hub::Hub;
use super::pages::{BodyTemplate, display_of, message};
use crate::range::today_date;
use crate::source::Source;
use crate::view::View;
use askama::Template;
use axum::extract::{Form, Query, State};
use axum::http::{HeaderMap, StatusCode, header};
use axum::response::sse::{Event, KeepAlive, Sse};
use axum::response::{IntoResponse, Redirect, Response};
use chrono::Utc;
use futures::StreamExt;
use std::collections::BTreeMap;
use std::convert::Infallible;
use std::sync::Arc;
use std::time::Duration;
use tokio::time::Instant;
use tokio_stream::wrappers::BroadcastStream;

pub const SLOW: Duration = Duration::from_secs(30);
pub const FAST: Duration = Duration::from_secs(5);
pub const FAST_FOR: Duration = Duration::from_secs(30 * 60);
pub const FAST_OFF: &str = "tempo real desligado depois de 30 min: atualiza a cada 30 s";

impl<S: Source> Hub<S> {
    /// A page starts listening; the poller wakes for it at once.
    pub fn subscribe(&self) -> tokio::sync::broadcast::Receiver<()> {
        let listener = self.tick.subscribe();
        self.wake.notify_one();
        listener
    }

    pub async fn interval(&self, now: Instant) -> Duration {
        if self.store.read().await.fast_until.is_some_and(|end| end > now) { FAST } else { SLOW }
    }

    /// The 5-second mode on for 30 minutes, or off; returns whether it is on.
    pub async fn toggle_fast(&self, now: Instant) -> bool {
        let on = {
            let mut store = self.store.write().await;
            store.fast_until = match store.fast_until {
                Some(end) if end > now => None,
                _ => Some(now + FAST_FOR),
            };
            store.notice = None;
            store.fast_until.is_some()
        };
        self.wake.notify_one();
        on
    }

    async fn expire_fast(&self, now: Instant) {
        let mut store = self.store.write().await;
        if store.fast_until.is_some_and(|end| end <= now) {
            store.fast_until = None;
            store.notice = Some(FAST_OFF.to_owned());
        }
    }

    /// The poller: idle while nobody listens; otherwise refresh, save when
    /// due and wait for the interval, a new listener or the fast switch.
    pub async fn run(self: Arc<Self>) {
        loop {
            while self.tick.receiver_count() == 0 {
                self.wake.notified().await;
            }
            self.expire_fast(Instant::now()).await;
            let now = Utc::now();
            self.refresh(now).await;
            self.save(now, false).await;
            let wait = self.interval(Instant::now()).await;
            tokio::select! {
                _ = tokio::time::sleep(wait) => {}
                _ = self.wake.notified() => {}
            }
        }
    }
}

async fn render_body<S: Source>(state: &AppState<S>, params: &BTreeMap<String, String>) -> String {
    // Parsed again on every tick, so "today" moves on at midnight.
    let Ok(view) = View::parse(params, today_date(Utc::now())) else {
        return String::new();
    };
    let display = display_of(state, &view).await;
    BodyTemplate { d: &display }
        .render()
        .unwrap_or_else(|_| r#"<p class="alert" role="alert">erro ao montar a página</p>"#.to_owned())
}

pub async fn events<S: Source>(
    State(state): State<AppState<S>>,
    Query(params): Query<BTreeMap<String, String>>,
) -> Response {
    if let Err(error) = View::parse(&params, today_date(Utc::now())) {
        return message(StatusCode::BAD_REQUEST, "Pedido inválido", &error);
    }
    let ticks = BroadcastStream::new(state.hub.subscribe()).map(|_| ());
    let stream = futures::stream::once(async {}).chain(ticks).then(move |()| {
        let state = state.clone();
        let params = params.clone();
        async move {
            let html = render_body(&state, &params).await;
            Ok::<Event, Infallible>(Event::default().event("view").data(html))
        }
    });
    Sse::new(stream).keep_alive(KeepAlive::default()).into_response()
}

pub async fn fast<S: Source>(
    State(state): State<AppState<S>>,
    headers: HeaderMap,
    Form(form): Form<BTreeMap<String, String>>,
) -> Response {
    let origin = headers.get(header::ORIGIN).and_then(|value| value.to_str().ok());
    if !same_origin(origin, &state.hosts) {
        return message(StatusCode::FORBIDDEN, "Origem não permitida", "o pedido veio de outro site");
    }
    state.hub.toggle_fast(Instant::now()).await;
```

```rust
pub fn bind_error(error: &std::io::Error, port: u16) -> String {
    if error.kind() == std::io::ErrorKind::AddrInUse {
        format!("porta {port} em uso; use --port")
    } else {
        format!("não consegui escutar na porta {port}: {error}")
    }
}

/// REQ-003, REQ-004: the key GOOGLE_APPLICATION_CREDENTIALS points at, gcloud
/// without it, and a readable reason when the file cannot be read.
pub fn token_source(credentials: Option<OsString>) -> TokenSource {
    match credentials {
        None => TokenSource::Gcloud,
        Some(path) => match std::fs::read_to_string(&path) {
            Ok(json) => TokenSource::choose(Some(&json)),
            Err(error) => TokenSource::Broken(format!("não consegui ler {}: {error}", path.to_string_lossy())),
        },
    }
}

pub fn open_browser(url: &str) {
    let program = if cfg!(target_os = "macos") { "open" } else { "xdg-open" };
    let _ = std::process::Command::new(program)
        .arg(url)
        .stdout(std::process::Stdio::null())
        .stderr(std::process::Stdio::null())
        .spawn();
}

pub async fn run(args: WebArgs, project: &str, no_cache: bool) -> anyhow::Result<()> {
    let now = Utc::now();
    let listener = TcpListener::bind((args.bind, args.port))
        .await
        .map_err(|error| anyhow::anyhow!(bind_error(&error, args.port)))?;
    if !args.bind.is_loopback() {
        eprintln!("aviso: escutando em {}; fora de um container, isso expõe o painel à rede", args.bind);
    }
    let hosts = AllowedHosts::new(args.host_name.as_deref());
    let open = (!args.no_open).then(|| format!("http://127.0.0.1:{}/", args.port));
    let shutdown = async {
        let _ = tokio::signal::ctrl_c().await;
    };
    if args.demo {
        let hub = Hub::new(Demo::new(DemoWorld::new(now)), EventCache::open(None, ""), DailyCache::open(None));
        return serve(Arc::new(hub), hosts, listener, open, shutdown).await;
    }
    let events_path = if no_cache {
        None
    } else {
        cache_path(std::env::var_os("XDG_CACHE_HOME"), std::env::var_os("HOME"), project)
    };
    let dailies = DailyCache::open(events_path.as_deref().map(dailies_path));
    let events = EventCache::open(events_path, &load_start(now));
    let tokens = Tokens::new(token_source(std::env::var_os("GOOGLE_APPLICATION_CREDENTIALS")), reqwest::Client::new());
    let hub = Hub::new(Remote::new(project, tokens), events, dailies);
    serve(Arc::new(hub), hosts, listener, open, shutdown).await
}

pub async fn serve<S: Source>(
    hub: Arc<Hub<S>>,
    hosts: AllowedHosts,
    listener: TcpListener,
    open: Option<String>,
    shutdown: impl Future<Output = ()> + Send + 'static,
) -> anyhow::Result<()> {
    let address = listener.local_addr()?;
    tokio::spawn(hub.clone().run());
    println!("stats web em http://{address}/");
    if let Some(url) = open {
        open_browser(&url);
    }
    let (told, mut heard) = tokio::sync::watch::channel(false);
    let server = axum::serve(listener, app(AppState { hub: hub.clone(), hosts })).with_graceful_shutdown(async move {
        shutdown.await;
        let _ = told.send(true);
    });
    tokio::select! {
        result = server => result?,
        _ = async {
            let _ = heard.wait_for(|stop| *stop).await;
            tokio::time::sleep(DRAIN).await;
        } => {}
    }
    hub.save(Utc::now(), true).await;
    Ok(())
}
```

```bash
#!/usr/bin/env bash
# Publishes the analytics panel to the homelab as http://stats.home, reachable
# only through Tailscale (docs/specs/stats-web REQ-021):
#   1. copies tools/stats and deploy/stats to the homelab, nothing else;
#   2. refuses to go on without the service account key in mode 600;
#   3. builds and (re)starts the container, published on loopback only;
#   4. adds a Caddy site that answers 403 outside the tailnet;
#   5. points stats.home at the homelab's Tailscale IP in Pi-hole;
#   6. proves it: 200 through Tailscale, 403 from the home network.
# Run it at home with Tailscale up: the 403 check needs the home network.
# Ends with STATS DEPLOY PASSED; that line, not the exit code, is the evidence.
# Env: STATS_HOST (ssh host, default homelab), STATS_PORT (default 7879).
# No `set -u`: the Bash tool's shell snapshot references unbound vars.
set -eo pipefail

ROOT="$(cd "$(dirname "${BASH_SOURCE[0]}")/.." && pwd)"
HOST="${STATS_HOST:-homelab}"
PORT="${STATS_PORT:-7879}"
DOMAIN="stats.home"
DIR="stats-web"
SSH_OPTS=(-o ServerAliveInterval=15 -o ServerAliveCountMax=8)

echo "==> syncing tools/stats and deploy/stats to $HOST:~/$DIR"
ssh "${SSH_OPTS[@]}" "$HOST" "mkdir -p $DIR/tools/stats $DIR/deploy/stats"
rsync -az --delete --exclude 'target' -e "ssh ${SSH_OPTS[*]}" "$ROOT/tools/stats/" "$HOST:$DIR/tools/stats/"
rsync -az --delete -e "ssh ${SSH_OPTS[*]}" "$ROOT/deploy/stats/" "$HOST:$DIR/deploy/stats/"
printf 'tools/stats/target\n' | ssh "${SSH_OPTS[@]}" "$HOST" "cat > $DIR/.dockerignore"

echo "==> building and starting the container, wiring Caddy and Pi-hole"
REMOTE_OUT="$(ssh "${SSH_OPTS[@]}" "$HOST" bash -s -- "$DIR" "$PORT" "$DOMAIN" <<'REMOTE'
set -eo pipefail
DIR="$1"; PORT="$2"; DOMAIN="$3"
cd "$HOME/$DIR"

KEY="$HOME/.config/stats/sa.json"
[ -f "$KEY" ] || { echo "FAIL: missing $KEY (tools/stats/README.md, Painel web no homelab)"; exit 1; }
[ "$(stat -c %a "$KEY")" = "600" ] || { echo "FAIL: $KEY must be mode 600"; exit 1; }

STATS_UID="$(id -u)" STATS_PORT="$PORT" docker compose -f deploy/stats/compose.yml up -d --build --remove-orphans >&2
docker image prune -f >/dev/null

TS_IP="$(tailscale ip -4 | head -1)"
[ -n "$TS_IP" ] || { echo "FAIL: the homelab has no Tailscale IPv4"; exit 1; }

# Caddy: append the site block once, then reload. Appending (never rewriting)
# keeps the other sites untouched and the bind-mounted file's inode stable.
CADDYFILE="$HOME/caddy/Caddyfile"
if ! grep -q "^http://$DOMAIN {" "$CADDYFILE"; then
  printf '\nhttp://%s {\n\t@outside not remote_ip 100.64.0.0/10 fd7a:115c:a1e0::/48\n\trespond @outside 403\n\theader X-Robots-Tag "noindex, nofollow"\n\treverse_proxy localhost:%s\n}\n' "$DOMAIN" "$PORT" >> "$CADDYFILE"
  echo "caddy: added $DOMAIN" >&2
fi
docker exec caddy caddy reload --config /etc/caddy/Caddyfile --adapter caddyfile >&2

# Pi-hole: "<tailscale ip> stats.home" once. The home network IP the other
# .home names use is kept for the 403 check below.
CURRENT="$(docker exec pihole pihole-FTL --config dns.hosts)"
LAN_IP="$(printf '%s' "$CURRENT" | grep -oE '([0-9]{1,3}\.){3}[0-9]{1,3}' | grep -v '^100\.' | head -1)"
if ! printf '%s' "$CURRENT" | grep -q " $DOMAIN\b"; then
  ITEMS="$(printf '%s' "$CURRENT" | sed -e 's/^\[ *//' -e 's/ *\]$//')"
  JSON="$(printf '%s' "$ITEMS" | awk -v add="$TS_IP $DOMAIN" 'BEGIN { RS = ", "; ORS = "" } NF { gsub(/\n/, ""); printf "%s\"%s\"", (n++ ? "," : ""), $0 } END { printf "%s\"%s\"", (n ? "," : ""), add }')"
  docker exec pihole pihole-FTL --config dns.hosts "[$JSON]" >/dev/null
  echo "pihole: added $TS_IP $DOMAIN" >&2
fi

for i in $(seq 1 90); do
  CODE="$(curl -s -o /dev/null -w '%{http_code}' -H "Host: $DOMAIN" "http://localhost:$PORT/healthz" || true)"
  [ "$CODE" = "200" ] && break
  sleep 2
done
[ "$CODE" = "200" ] || { echo "FAIL: the container did not answer 200 on :$PORT"; docker logs --tail 30 stats-web >&2; exit 1; }
echo "TS_IP=$TS_IP"
echo "LAN_IP=$LAN_IP"
REMOTE
)" || { printf '%s\n' "$REMOTE_OUT"; exit 1; }

TS_IP="$(printf '%s\n' "$REMOTE_OUT" | sed -n 's/^TS_IP=//p')"
LAN_IP="$(printf '%s\n' "$REMOTE_OUT" | sed -n 's/^LAN_IP=//p')"
[ -n "$TS_IP" ] && [ -n "$LAN_IP" ] || { echo "FAIL: could not read the homelab addresses"; printf '%s\n' "$REMOTE_OUT"; exit 1; }

echo "==> checking http://$DOMAIN through Tailscale ($TS_IP)"
CODE="$(curl -s -o /dev/null -w '%{http_code}' --max-time 15 --resolve "$DOMAIN:80:$TS_IP" "http://$DOMAIN/healthz" || true)"
[ "$CODE" = "200" ] || { echo "FAIL: http://$DOMAIN answered $CODE through Tailscale (is Tailscale up here?)"; exit 1; }

echo "==> checking http://$DOMAIN from the home network ($LAN_IP)"
CODE="$(curl -s -o /dev/null -w '%{http_code}' --max-time 15 --resolve "$DOMAIN:80:$LAN_IP" "http://$DOMAIN/healthz" || true)"
[ "$CODE" = "403" ] || { echo "FAIL: http://$DOMAIN answered $CODE from the home network; it must be 403"; exit 1; }

RESOLVED="$(getent hosts "$DOMAIN" | awk '{ print $1 }' | head -1 || true)"
[ "$RESOLVED" = "$TS_IP" ] || echo "note: $DOMAIN resolves to '${RESOLVED:-nothing}' here, not $TS_IP yet (DNS cache, or the split DNS is not set)"

echo "STATS DEPLOY PASSED"
```
