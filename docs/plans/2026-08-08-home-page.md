`src/lib/home/services.ts`:

```ts
interface Service {
  title: string;
  description: string;
  caseSlug: string;
}

/**
 * F1.4: every advertised service must reference a case that proves it.
 * The profile schema guarantees caseSlug is non-empty, but only here do the
 * two collections meet, so only here can the reference be resolved.
 */
export const assertServicesHaveCases = (
  services: readonly Service[],
  knownSlugs: readonly string[],
): void => {
  const known = new Set(knownSlugs);
  const orphans = services.filter((service) => !known.has(service.caseSlug));

  if (orphans.length > 0) {
    const detail = orphans
      .map((service) => `"${service.title}" -> ${service.caseSlug}`)
      .join('; ');

    throw new Error(
      `Every advertised service needs a case that proves it. Missing: ${detail}`,
    );
  }
};
```

```ts
import type { Profile } from '../../content/schemas/profile';

export interface PersonLd {
  '@context': 'https://schema.org';
  '@type': 'Person';
  name: string;
  jobTitle: string;
  description: string;
  url: string;
  knowsAbout: string[];
  sameAs: string[];
}

const PUBLIC_PROFILE_KINDS = new Set(['github', 'linkedin', 'site']);

export const personLd = (profile: Profile, site: string): PersonLd => ({
  '@context': 'https://schema.org',
  '@type': 'Person',
  name: profile.name,
  jobTitle: profile.headline,
  description: profile.bio,
  url: site,
  knowsAbout: [...profile.stack],
  sameAs: profile.contacts
    .filter((contact) => PUBLIC_PROFILE_KINDS.has(contact.kind))
    .map((contact) => contact.value),
});
```
